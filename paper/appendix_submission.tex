\section{Proofs for Generalized Reparameterization and Closed-Loop Filtering}
\label{app:proofs}

Throughout this section, all regimes share common physical transition and reward maps, a frozen deterministic canonical policy $\pi_r:\mathcal{S}\to\mathcal{A}=[-1,1]^d$, and known invertible actuator transforms $\{\gain_z\}_{z=0}^{M-1}$.

\subsection{Box-Constrained Projection, Feasibility, and Pathwise Equivalence}

For an invertible matrix $\gain_z$, put $w=\gain_z^{-1}v^\star$ with $v^\star=\gain_r q$. Left-multiplying $\gain_z\clip(w,-1,1)=v^\star$ by $\gain_z^{-1}$ gives $\clip(w,-1,1)=w$, whose fixed-point set is $\mathcal{A}$, establishing Eq.~\eqref{eq:compensation_feasibility}.

\begin{proofof}{Proposition~\ref{prop:box_projection}}
For any scalar $x\in\mathbb{R}$ and $a\in[-1,1]$, case analysis on $x<-1$, $|x|\le 1$, and $x>1$ shows $|{\clip(x,-1,1)-x}|=\max(0,|x|-1)\le |a-x|$. Setting $x_i=v_i^\star/g_{z,i}$ ($g_{z,i}\ne 0$), we have $g_{z,i}a_i-v_i^\star=g_{z,i}(a_i-x_i)$. Squaring and multiplying by $w_i g_{z,i}^2\ge 0$ for each coordinate $i\in\{1,\ldots,d\}$ yields
\begin{equation}
 w_i\left(g_{z,i}\clip(x_i,-1,1)-v_i^\star\right)^2
 \le w_i\left(g_{z,i}a_i-v_i^\star\right)^2,
\end{equation}
and summing over $i$ proves Eq.~\eqref{eq:weighted_projection}. Taking $w_i=1$ gives $|g_{z,i}\clip(v_i^\star/g_{z,i},-1,1)-v_i^\star|=|g_{z,i}|\max(0,|v_i^\star|/|g_{z,i}|-1)=\max(0,|v_i^\star|-|g_{z,i}|)$, proving Eq.~\eqref{eq:infeasibility_residual}, which equals zero whenever $|v_i^\star|\le |g_{z,i}|$ for all $i$.
\end{proofof}

\begin{proofof}{Proposition~\ref{prop:equivalence}}
Couple the native reference-regime rollout (fixed in mode $r$) and the switching rollout (following $(z_t)_{t=0}^{H-1}$) by sharing the initial simulator state $x_0$ and exogenous randomness $(\varepsilon_t)_{t=0}^{H-1}$. When $\gain_{z_t}(\mathcal{A})=\mathcal{A}$ (including diagonal sign matrices $\gain_{z_t}^{-1}=\gain_{z_t}$ and signed-permutation matrices $\gain_{z_t}^{-1}=\gain_{z_t}^\top$), $\gain_{z_t}^{-1}\gain_r\pi_r(s_t)\in\mathcal{A}$ holds at shared state $x_t$, so clipping in Eq.~\eqref{eq:compensation_general} is inactive and $\gain_{z_t}a_t^{\mathrm{comp}}=\gain_r\pi_r(s_t)$. Shared noise and clipping yield identical executed actions $u_t$, next states $x_{t+1}$, and rewards $r_t$; induction over $t=0,\ldots,H-1$ establishes pathwise and distributional equality.
\end{proofof}

\subsection{Action-Weighted Mode Error and Channel-Factored Exactness}

\begin{proofof}{Proposition~\ref{prop:mode_error}}
At common state $x_t$ with $q_t=\pi_r(s_t)$, coordinate $i$ of $e_t=\gain_{z_t}\gain_{\hat z_t}\gain_r q_t-\gain_r q_t$ equals $(g_{z_t,i}g_{\hat z_t,i}-1)g_{r,i}q_{t,i}$, which is $0$ if $g_{z_t,i}=g_{\hat z_t,i}$ and $\pm 2 q_{t,i}$ if $g_{z_t,i}\ne g_{\hat z_t,i}$, yielding Eq.~\eqref{eq:weighted_mode_error}. Because scalar clipping is $1$-Lipschitz, $\|\clip(v_t^C+\varepsilon_t,-1,1)-\clip(v_t^O+\varepsilon_t,-1,1)\|_2\le \|e_t\|_2$.
\end{proofof}

\begin{proofof}{Proposition~\ref{prop:factored_exact}}
For any $g\in\{-1,+1\}^d$, $\phi_i(g_i)\le \max_{s\in\{-1,+1\}}\phi_i(s)=\phi_i(\hat g_i)$ for every $i$, so summing over $i=1,\ldots,d$ gives $\Phi(g)\le\Phi(\hat g)$. If $\phi_i(g_i^\star)>\phi_i(-g_i^\star)$ for all $i$, then $\hat g_i=g_i^\star$ coordinatewise, and any $g\ne g^\star$ differs on at least one index $i_0$, yielding $\Phi(g)<\Phi(g^\star)$.
\end{proofof}

\subsection{Causal Execution and Mode Distinguishability}
\label{app:causal_identification}

Before selecting $a_t$, the pre-action $\sigma$-algebra available to the agent is $\mathcal{H}_t=\sigma(s_0,(a_\tau,r_\tau,s_{\tau+1})_{\tau=0}^{t-1})$. Because $\pi_r$, $\{f_j\}_{j=1}^K$, emission scoring, and both filter updates~\eqref{eq:posterior}--\eqref{eq:factored_filter} are deterministic Borel-measurable maps, $(b_t,s_t)$ determines $\hat z_t$ and $a_t$ prior to step-$t$ execution, while $(s_t,a_t,s_{t+1})$ updates $b_{t+1}$ only afterward; induction confirms $\mathcal{H}_t$-measurability of $a_t$. Conversely, if two modes $z\ne z'$ satisfy $\clip(\gain_z a_t,-1,1)=\clip(\gain_{z'} a_t,-1,1)$ along all reachable histories under a mode-blind $\mathcal{H}_t$-measurable policy, they induce identical trajectory laws $P_z=P_{z'}$, capping equal-prior binary classification accuracy at $1/2$.

\subsection{Generalized Return Decomposition, Filter Recovery, and Lean 4 Formalization}
\label{app:return_proof}

\begin{proofof}{Theorem~\ref{thm:return_gap}}
Linearity of expectation under integrable finite-horizon returns gives $J_C-J_{\mathrm{ref}}=(J_P-J_{\mathrm{ref}})+(J_C-J_P)$. Define projected-oracle values backward from $V_H^P(x)=0$ via $Q_t^P(x,v)=\mathbb{E}[r_t+V_{t+1}^P(x_{t+1})\mid x_t=x,\,u_t=\clip(v+\varepsilon_t,-1,1)]$ and $V_t^P(x)=Q_t^P(x,v_t^P(x))$. Because $P$ depends only on physical state $x_t$, conditioning step $t$ of causal rollout $C$ on $(x_t,b_t)$ gives $\mathbb{E}_C[Q_t^P(x_t,v_t^C)-V_t^P(x_t)]=\mathbb{E}_C[r_t+V_{t+1}^P(x_{t+1})-V_t^P(x_t)]$. Summing over $t=0,\ldots,H-1$ telescopes $\sum_{t=0}^{H-1}\mathbb{E}_C[V_{t+1}^P(x_{t+1})-V_t^P(x_t)]=-\mathbb{E}[V_0^P(x_0)]=-J_P$~\citep{Kakade2002PDL}, yielding Eq.~\eqref{eq:causal_return_identity}. When $\gain_{z_t}$ is box-preserving, Proposition~\ref{prop:equivalence} gives $J_P=J_O=J_{\mathrm{ref}}$, and substituting $|Q_t^O(x_t,v_t^C)-Q_t^O(x_t,v_t^O)|\le L_t^{\mathrm{loc}}(x_t)\|v_t^C-v_t^O\|_2$ together with Proposition~\ref{prop:mode_error} yields Eq.~\eqref{eq:causal_return_bound}.
\end{proofof}

\begin{proofof}{Theorem~\ref{thm:filter_recovery}}
For Clause~1, when $\delta=1$ and $h=0$, log-odds evolve additively as $L_{t+1}=L_t+s_t$; $T_{\mathrm{dwell}}$ steps of $+m$ yield $L_{T_{\mathrm{dwell}}}=L_0+T_{\mathrm{dwell}}m$, so $N$ opposing steps of $-m'$ give $L_0+T_{\mathrm{dwell}}m-Nm'<0\iff N>(L_0+T_{\mathrm{dwell}}m)/m'$. When $0\le\delta<1$ and $|s_t|\le S$, $|L_0|\le S/(1-\delta)$ implies $|L_{t+1}|\le \delta|L_t|+|s_t|\le \delta S/(1-\delta)+S=S/(1-\delta)$ by induction on $t$.

For Clause~2, if $|\hat v_{t_0+\tau,i}^{(j)}-\tilde v_{t_0+\tau,i}(z)|\le \epsilon_i$ for all $j,i$, expanding $(\tilde v(w)-\hat v^{(j)})^2-(\tilde v(z)-\hat v^{(j)})^2$ bounds every head's log-density difference below by $\Gamma_{t_0+\tau}(z,w;\epsilon)$, which is preserved under log-mean-exp aggregation (\texttt{inverseEvidence\_signal\_margin}). Because sticky mixing pulls positive log-odds toward zero ($\log(\bar b(z)/\bar b(w))\ge \min(\log(b(z)/b(w)),0)$; \texttt{stickyPrior\_\allowbreak log\_\allowbreak odds\_\allowbreak lower}) while $b_{t_0}\in\Delta^{M-1}$ imposes initial floor $\log(\bar b_{t_0}(z)/\bar b_{t_0}(w))\ge -\log\frac{1-h+h/(M-1)}{h/(M-1)}$, induction on Eq.~\eqref{eq:posterior} over $\tau=0,\ldots,N-1$ yields $\log(b_{t_0+N}(z)/b_{t_0+N}(w))\ge \min\!\left(c\,m,\;c\,m\sum_{\tau=0}^{N-1}\delta^\tau-\delta^N\log\frac{1-h+h/(M-1)}{h/(M-1)}\right)>0$ for every $w\ne z$ (\texttt{temperedFilter\_\allowbreak margin\_\allowbreak envelope}). Applying Markov's inequality $\Pr(|\hat v_{\tau,i}^{(j)}-\tilde v_{\tau,i}(z)|>\epsilon_i)\le M_{\tau,j,i}/\epsilon_i^2$ and a finite union bound over $(\tau,j,i)$ plus the under-excitation event completes the proof.
\end{proofof}

\paragraph{15-module Lean 4 formalization.}
The standalone Lean 4 package (\texttt{reproducibility/\allowbreak lean/}, pinned to Lean/Mathlib 4.28.0 across 15 environment-independent modules with zero \texttt{sorry} axioms) mechanizes:
\begin{itemize}[leftmargin=*]
    \item \textbf{Lossy projection \& factored hypercubes (\texttt{LossyProjection\allowbreak AndFactored\allowbreak Filter.lean}):} Proposition~\ref{prop:box_projection} (\texttt{weightedBoxProjection\_\allowbreak optimal}, \texttt{lossyProjection\_\allowbreak infeasibility\_\allowbreak residual}, \texttt{lossyProjection\_\allowbreak zero\_\allowbreak of\_\allowbreak sufficient\_\allowbreak gain}), the two-term decomposition (\texttt{lossy\_\allowbreak two\_\allowbreak term\_\allowbreak return\_\allowbreak decomposition}), Theorem~\ref{thm:filter_recovery}(1) (\texttt{untempered\_\allowbreak switch\_\allowbreak delay\_\allowbreak scales\_\allowbreak with\_\allowbreak dwell}, \texttt{tempered\_\allowbreak log\_\allowbreak odds\_\allowbreak uniform\_\allowbreak cap}), and Proposition~\ref{prop:factored_exact} (\texttt{channelFactored\_\allowbreak map\_\allowbreak maximizes\_\allowbreak joint}, \texttt{channelFactored\_\allowbreak map\_\allowbreak unique\_\allowbreak recovery}).
    \item \textbf{Closed-loop filter stability \& recovery (\texttt{ActionCompensation\allowbreak Filter.lean}, \texttt{ActionCompensation\allowbreak Evidence.lean}, \texttt{ActionCompensation\allowbreak Recovery.lean}):} odds-ratio nonexpansiveness (\texttt{stickyPrior\_\allowbreak log\_\allowbreak stability}), uniform log-odds error floor $2c\eta/(1-\delta)$ (\texttt{temperedFilter\_\allowbreak error\_\allowbreak floor}), excitation margin (\texttt{inverseEvidence\_\allowbreak signal\_\allowbreak margin}), and Theorem~\ref{thm:filter_recovery}(2) (\texttt{inverse\_\allowbreak filter\_\allowbreak recovery\_\allowbreak probability}).

\section{Estimator Details and Offline Budget Accounting}
\label{app:estimator}

The frozen HalfCheetah inverse-dynamics ensemble comprises $K=5$ independently parameterized heads. Each head is a three-hidden-layer MLP ($51\to 256\to 256\to 256\to 6$) with SiLU hidden activations and a final $\tanh$ layer predicting the nominal pre-noise signal $v_t=\clip(\gain_{z_t}a_t,-1,1)$. Given consecutive 17-dimensional observations $(s_t,s_{t+1})$, the 51-dimensional input feature vector is
\begin{equation}
    \xi_t=\left[
      \tanh(s_t/5),\;
      \tanh(s_{t+1}/5),\;
      \tanh((s_{t+1}-s_t)/0.1)
    \right].
\end{equation}

The released estimator (\texttt{model\_seed = 20260732}) was fitted in two offline stages:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Stage 1 (initialization, 3,000 updates).} Trained from scratch with learning rate $3\times 10^{-4}$ on 96,000 labeled transitions (64,000 stationary; 32,000 switching; event seeds 92001 and 92002) rolled out from robust and true-mode-conditioned controllers on seeds 8 and 16, and validated on 48,000 separate transitions from controller seed 24 (event seeds 93001 and 93002).
    \item \textbf{Stage 2 (warm-started fine-tuning, 1,500 updates).} Initialized from Stage 1 and fine-tuned at learning rate $10^{-4}$ on 82,000 labeled training transitions (event seeds 154001 and 154013) collected across robust SAC, four fixed-mode specialists, dynamic-oracle, and causal-router rollouts from source seed 4021, with 82,000 separate validation transitions collected from disjoint source seed 4049 (event seeds 154101 and 154113).
\end{enumerate}
Both stages used AdamW (weight decay $10^{-5}$), global gradient-norm clipping at 10.0, and independent bootstrap minibatches of 256 transitions per head. Across both stages, estimator fitting consumed $96{,}000+82{,}000=178{,}000$ supervised training transitions and $48{,}000+82{,}000=130{,}000$ validation transitions.

\paragraph{Upstream trajectory-provider budget.}
To make offline data accounting fully transparent, the manifest (\texttt{reproducibility/v32/estimator/model\_manifest.json}) records 6 Stage-1 provider policies (4 training on seeds 8 and 16; 2 validation on seed 24) and 10 Stage-2 provider policies (robust SAC plus four specialists on training seed 4021 and validation seed 4049). Each historical provider was trained for 5.6M environment steps, totaling $9\times 5.6\text{M}=50.4\text{M}$ upstream steps for training-trajectory providers and $7\times 5.6\text{M}=39.2\text{M}$ steps for validation-trajectory providers ($89.6\text{M}$ upstream steps overall). All provider, calibration, and validation seeds precede and are strictly disjoint from the pilot and prospective confirmation seeds; the estimator was frozen prior to registration and never refitted.

Empirical inverse-residual variances $\sigma_{\mathrm{inv}}^2$ are computed on the Stage-2 training split and clipped to $[10^{-4},0.25]$, yielding
\begin{equation}
 \sigma_{\mathrm{inv}}^2 = (0.1289,\;0.1030,\;0.1174,\;0.1357,\;0.1063,\;0.0967).
\end{equation}
Filter hyperparameters $(h,c,\delta)=(0.002,1.0,0.98)$ were selected on the disjoint validation splits and held fixed across all HalfCheetah and Reacher deployments.

\section{Prospective Training and Evaluation Protocol}
\label{app:protocol}

\subsection{Policy Budgets}

For each seed in the canonical BAPR path, a multimode SAC source controller trains for 1,400 iterations of 4,000 environment interactions and 250 gradient updates (5.6M interactions; 350,000 updates). Its actor, twin critics, target critics, and log-temperature are then forked into fixed reference mode 0 for 700 additional iterations (2.8M interactions; 175,000 updates), resetting replay-buffer contents and Adam states at the fork boundary. The resulting canonical controller receives 8.4M interactions and 525,000 gradient updates, matching SAC, recurrent ESCP, and \resacbaseline{} step-for-step in online controller training.

\subsection{Comparator Architectures and Hyperparameters}

All learned HalfCheetah arms share discount $\gamma=0.99$, target Polyak rate $\tau=0.005$, replay capacity $10^6$, and batch size 256:
\begin{itemize}[leftmargin=*]
    \item \textbf{SAC and canonical BAPR controller.} Both use two-hidden-layer width-256 ReLU actors ($\tanh$-squashed Gaussian), three-hidden-layer width-256 ReLU twin critics, Adam learning rate $3\times 10^{-4}$, automatic entropy tuning toward $-\dim(\mathcal{A})$, twin-minimum Bellman backups, and twin-mean actor Q-objectives. Standalone SAC/ESCP/\resacbaseline{} runs collect 4,000 uniform warmup steps within their 8.4M budget, whereas the canonical two-stage trainer starts policy sampling immediately.
    \item \textbf{\resacbaseline{}.} Uses a five-critic width-256 ReLU ensemble with target blend $0.75$ per-head + $0.25$ ensemble-minimum, critic LCB annealed from $-2.0$ to $0$ after a 20\% warmup, actor anchor weight $10^{-3}$, evaluation EMA rate $0.005$, and disabled conservative shift/OOD/BC terms ($\texttt{weight\_reg}=0$, $\texttt{beta\_ood}=0$, $\texttt{beta\_bc}=0$).
    \item \textbf{Recurrent ESCP.} Follows \citet{Luo2022ESCP} (\texttt{jax\_experiments/\allowbreak networks/\allowbreak escp\_recurrent.py}): a 16-step state--action history feeds a recurrent context encoder (\texttt{FC(128)} with LeakyReLU $\to$ \texttt{GRU(64)} $\to$ \texttt{FC(2)} with $\tanh$), conditioning \texttt{[128, 64]} LeakyReLU actor and twin-critic MLPs with twin-minimum actor and critic targets, actor/encoder learning rate $3\times 10^{-4}$, critic learning rate $10^{-3}$, temperature learning rate $10^{-2}$ (capped at $1.0$, target-entropy ratio $1.5$). During training, ESCP receives privileged true regime IDs to optimize its RMDM auxiliary contrast and task prototypes once 100,000 steps and at least two regimes have been observed; evaluation uses zero privileged labels.
\end{itemize}

\subsection{Decomposition of the Composite Gate Outcome and Switch-Delay Definitions}

BAPR achieves positive paired 95\% confidence intervals against all three equal-budget comparators and passes every consistency gate against ESCP and \resacbaseline{}, but misses the composite gate against SAC on seed wins (7/10 vs.\ 8/10), event wins (23/30 vs.\ 24/30), and $\ge 10\%$ oracle-headroom count (7/10 vs.\ 8/10). On the three SAC non-winning seeds (88021, 88039, 88179), BAPR minus SAC equals $-45.1$, $-228.0$, and $-1.2$, corresponding to oracle headroom over SAC of only $+2.3\%$, $-2.3\%$, and $+4.1\%$.

The compensation-recovery ratio
\begin{equation}
 R_{\mathrm{comp}}=
 \frac{J_{\mathrm{causal}}-J_{\mathrm{no\text{-}comp}}}
 {J_{\mathrm{oracle}}-J_{\mathrm{no\text{-}comp}}}
\end{equation}
reaches $95.9\%$ at the group means, where oracle headroom over SAC is $1134.9$, inference loss is $125.0$, and net BAPR gain over SAC is $1009.9$. Across this paper we distinguish three complementary transient diagnostics:
\begin{enumerate}[leftmargin=*]
    \item \textbf{First-correct switch delay:} the earliest offset $\Delta\ge 0$ after a switch at $t_{\mathrm{sw}}$ where $\hat z_{t_{\mathrm{sw}}+\Delta}=z_{t_{\mathrm{sw}}}$.
    \item \textbf{Sustained (eight-step) switch delay:} the earliest offset $\Delta\ge 0$ where $\hat z_{t_{\mathrm{sw}}+\Delta+\tau}=z_{t_{\mathrm{sw}}}$ for eight consecutive steps $\tau\in\{0,\ldots,7\}$ (counting blocks without an eight-step run as unresolved).
    \item \textbf{Cumulative post-switch wrong-mode fraction over window $W$:} $\frac{1}{W}\sum_{\tau=0}^{W-1}\mathbb{I}[\hat z_{t_{\mathrm{sw}}+\tau}\ne z_{t_{\mathrm{sw}}}]$. Note that this is a \emph{cumulative window average} rather than the instantaneous error rate at step $W$: on the ten-seed Brax-Spring HalfCheetah confirmation (3-step median first-correct delay; 0/450 switches unresolved), cumulative wrong-mode averages over $W\in\{1,2,4,8\}$ are 99.8\%, 99.6\%, 82.7\%, and 46.8\% precisely because ~3.7 pre-recovery steps out of the first 8 steps precede mode lock ($3.74/8\approx 46.8\%$), whereas overall trajectory Brier score is 0.0246 and mean absolute nominal-signal error is 0.0104.
\end{enumerate}

\begin{equation}

\section{Standard Gymnasium MuJoCo v5 Multi-Morphology Protocol}
\label{app:mujoco_v5_protocol}

To test reference-only BAPR calibration in a different physics engine, with matched actor and critic hidden widths, we trained and audited five fresh seeds ($\{51003,51021,51039,51057,51079\}$) across four standard Gymnasium 1.2.0 / MuJoCo 3.2.7~\citep{Todorov2012MuJoCo,Towers2024Gymnasium} locomotion tasks under pinned PyTorch 2.5.1+cu121: \texttt{HalfCheetah-v5} ($s\in\mathbb{R}^{17},a\in[-1,1]^6$), \texttt{Ant-v5} with proprioceptive state (\texttt{include\_\allowbreak cfrc\_\allowbreak ext\_\allowbreak in\_\allowbreak observation=\allowbreak False}, $s\in\mathbb{R}^{27},a\in[-1,1]^8$), \texttt{Walker2d-v5} ($s\in\mathbb{R}^{17},a\in[-1,1]^6$), and \texttt{Hopper-v5} ($s\in\mathbb{R}^{11},a\in[-1,1]^3$). Every episode spans 1,000 steps with 250-step mode dwell intervals and $\sigma=0.02$ Gaussian actuator noise; on terminable morphologies (\texttt{Ant-v5}, \texttt{Walker2d-v5}, \texttt{Hopper-v5}), early physical termination ends the episode cleanly as in standard Gymnasium evaluation.

Each learner collects 500,000 transitions across eight vectorized environments with \texttt{SAME\_STEP} autoreset. Replay uses the terminal successor from \texttt{infos["final\_obs"]}. The warmup is 4,800 transitions, batch size 256, $\gamma=0.99$, $\tau=0.005$, and Adam learning rate $3\times10^{-4}$. The common actor widths are \texttt{[256, 256]} and twin-critic widths are \texttt{[256, 256, 256]}. All published runs started from scratch. The software release stores the simulator, replay, learner and RNG state together for future checkpoints.
\begin{itemize}[leftmargin=*]
    \item \textbf{Self-calibrated BAPR and burst-free control (zero external steps, zero mode labels):} trains strictly in fixed reference mode 0 for 500k steps, initiating 4-step uniform action bursts at per-step trigger probability $0.0125$ ($\approx 5\%$ exploratory steps). At step 500k, the five-head inverse ensemble ($K=5$, three width-256 SiLU layers) fits directly on a 90\%/10\% train/validation split of that 500k replay buffer for 4,500 AdamW updates using $v_t=\gain_0 a_t$. A matched \textbf{burst-free BAPR control} (\texttt{bapr\_noburst}) trains identically in mode 0 with $0\%$ uniform bursts, isolating the effect of burst exploration on both canonical headroom and self-calibrated identification. On the exact same 90\%/10\% mode-0 split and 4,500 updates, we also fit a \textbf{structure-matched five-head shared-physics forward ensemble} $g_\psi(s_t,v_t)\approx\tanh((s_{t+1}-s_t)/0.1)$ evaluated online at $\tilde v_t(z)=\clip(\gain_z a_t,-1,1)$.
    \item \textbf{Memoryless SAC:} trains across the four polarity regimes with 250-step within-episode switching.
    \item \textbf{ESCP-256 adaptation:} upgrades \citet{Luo2022ESCP} to match BAPR's \texttt{[256, 256]} / \texttt{[256, 256, 256]} actor--critic capacity alongside a 16-step state--action history encoder (\texttt{FC(256)} $\to$ \texttt{GRU(128)} $\to$ 8D $\tanh$ context) trained with an RMDM-style objective: EMA per-regime prototypes ($\tau_{\mathrm{proto}}=0.995$) for critic conditioning, within-regime consistency loss ($w_{\mathrm{cons}}=50.0$), and cross-regime element-wise RBF kernel log-determinant diversity ($w_{\mathrm{div}}=0.025$, $r_{\mathrm{rbf}}=80.0$, bottleneck noise $\sigma=0.01$). The port uses an element-wise RBF kernel and applies the diversity term when consistency is below $0.1$.
    \item \textbf{RMA adaptation~\citep{Kumar2021RMA}:} uses a SAC base learner and GRU adapter. It allocates 400k steps to Phase~1 (jointly training the \texttt{[256, 256]} base policy, twin critics, and privileged gain-vector encoder $E(\operatorname{diag}(\gain_{z_t}))\in\mathbb{R}^8$) and 100k steps to Phase~2 (training the online 16-step history adapter $\hat e_t=A(h_t)\in\mathbb{R}^8$ via MSE regression to $E(\operatorname{diag}(\gain_{z_t}))$ under on-policy adapter rollouts), totaling exactly 500k steps.
    \item \textbf{Belief-conditioned SAC:} collects 500k multimode transitions and conditions actor and critic on $[s_t,b_t]$. The frozen inverse ensemble and filter $(0.002,1,0.98)$ are shared with the paired BAPR run. The estimator was fitted on that run's additional 500k reference transitions, reused by both routes. This controls the supplied estimator, while training curricula and access to the analytic action map differ.
\end{itemize}

The sign-product decoder scores the entire Boolean action-sign space even when the evaluation environment uses four modes. Its independent binary priors and per-channel mixture evidence differ from those of the four-mode joint filter. In particular, the initial tie can select a sign vector outside the environment's four-mode set.

\section{Historical Brax Cross-Environment Diagnostic Boundary}
\label{app:cross_environment}

Prior to introducing self-calibrated single-mode training and standard MuJoCo v5 evaluation (Section~\ref{sec:mujoco_v5}), exploratory screens on Brax's \texttt{spring} backend used the two-stage multimode warm-start protocol:
\begin{itemize}[leftmargin=*]
    \item \textbf{Brax Ant, Hopper, and Walker2d.} Under Brax's \texttt{spring} contact model and multimode source training, a three-seed Ant true-mode audit showed large return headroom (4510.7 vs.\ 2588.6 for robust SAC) but only 1/3 seeds cleared the strict zero-termination gate, while Hopper and Walker2d terminated before step 250. Section~\ref{sec:mujoco_v5} resolves this external-validity question directly on standard MuJoCo v5 across all four morphologies.
    \item \textbf{Brax Swimmer.} Across three seeds (8.4M steps/actor, 250-step dwell, 1,000-step episodes), true-mode compensation matched native reference returns identically (50.30) but did not exceed memoryless robust SAC (55.04), illustrating that low-dimensional undulatory gaits without opposing ground-reaction impulses may admit competitive memoryless compromises.
    \item \textbf{Bus fleet control.} Stochastic bus corridors exhibit persistent demand and travel-time shifts rather than known invertible actuator-coordinate maps, motivating uncertainty-regularized ensemble critics~\citep{Zhang2026RESAC,Zhang2026SingleAgentBus} rather than action-space reparameterization.
\end{itemize}
